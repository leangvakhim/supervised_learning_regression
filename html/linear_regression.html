Python Implementation (scikit-learn)

# 1. Import tools: numpy (arrays) & scikit-learn (ML)
import numpy as np
from sklearn.linear_model import LinearRegression
# 2. Provide our Student Data
# X (Hours) must be a 2D array (column) for scikit-learn
X = np.array([[15], [25], [35], [45], [55], [65], [75], [85], [90]])
y = np.array([20, 35, 30, 50, 45, 65, 60, 85, 80])
# 3. Create the empty Machine Learning Model
model = LinearRegression()
# 4. 'Train' the model
# This runs all the math & formulas we just learned!
model.fit(X, y)
# 5. Make a prediction for a new student (80 hours)
new_student = np.array([[80]])
prediction = model.predict(new_student)
# 6. View the result
print(f"Predicted Score: {prediction[0]:.1f}%")
